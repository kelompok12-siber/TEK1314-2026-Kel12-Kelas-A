# /etc/systemd/system/bilicheck.service
[Unit]
Description=Bilicheck Flask Dashboard
After=network.target mysql.service
Requires=mysql.service

[Service]
Type=simple
User=kel-12
WorkingDirectory=/home/kel-12/bilicheck-backend
EnvironmentFile=/etc/bilicheck.env
ExecStart=/home/kel-12/bilicheck-backend/.venv/bin/python /home/kel-12/bilicheck-backend/app.py
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/bilicheck.service.d/hardening.conf
[Service]
NoNewPrivileges=yes
PrivateTmp=yes
ProtectSystem=full
ProtectKernelTunables=yes
ProtectControlGroups=yes
